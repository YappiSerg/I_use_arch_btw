#include <iostream>
using namespace std;

long long count(long long n) {
    if (n == 0) return 1;
    if (n == 1) return 2;

    long long a = 1, b = 2, c;
    for (int i = 2; i <= n; i++) {
        c = a + b;
        a = b;
        b = c;
    }
    return b;

}

int main() {
    cout << "n = ";
    long long a; cin >> a;

    cout << "amount of sequences: " << count(a) << endl;
    return 0;
}
